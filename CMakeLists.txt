cmake_minimum_required(VERSION 3.16)
project(pokshell_alpha C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
add_executable(pocket-session src/session.c)
target_link_libraries(pocket-session PRIVATE m)
add_executable(pocket-game src/game.c)
target_link_libraries(pocket-game PRIVATE m)
add_executable(pocketctl src/pocketctl.c)
install(TARGETS pocket-session pocket-game pocketctl RUNTIME DESTINATION bin)
